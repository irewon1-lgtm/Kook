const {chromium,expect}=require('/workspace/kr4-pc-web/web/node_modules/@playwright/test');
const {readFileSync,writeFileSync}=require('node:fs');
const {createHash}=require('node:crypto');
(async()=>{
const expected=JSON.parse(readFileSync('/workspace/kr4-pc-work/quarter-op-independent-expected.json','utf8'));
const browser=await chromium.launch({headless:true,executablePath:'/usr/bin/chromium'});
const context=await browser.newContext({viewport:{width:1365,height:853},locale:'ko-KR',timezoneId:'Asia/Seoul'});
const page=await context.newPage(),errors=[],checks=[];page.on('pageerror',e=>errors.push(String(e)));
for(const record of expected.records){
 await page.goto(`http://127.0.0.1:4173/Kook/#stock/${record.code}/2`);
 await expect(page.locator('.dtl-quarter-table')).toBeVisible();
 const metadata=await page.getByTestId('kr4-app').evaluate(e=>({generationId:e.dataset.generationId,engineCommit:e.dataset.sourceCommit,uiCommit:e.dataset.uiCommit,dataBase:e.dataset.dataBase}));
 const base=new URL(metadata.dataBase,page.url());const manifest=await (await page.request.get(new URL('manifest.json',base).href)).json();
 const response=await page.request.get(new URL(`details/${record.code}.json`,new URL(manifest.basePath||'',base)).href),bytes=await response.body();
 expect(response.ok()).toBe(true);expect(createHash('sha256').update(bytes).digest('hex')).toBe(record.sourceSha256);
 const rows=page.locator('.dtl-quarter-table tbody tr');expect(await rows.count()).toBe(record.rows.length);
 for(let i=0;i<record.rows.length;i++){
  const row=record.rows[i],cells=rows.nth(i).locator('th,td');expect((await cells.nth(0).innerText()).trim()).toBe(row.period);
  for(let j=0;j<2;j++){const actual=(await cells.nth(6+j).innerText()).trim(),value=row.columns[j];expect(actual).toBe(value.expected);checks.push({issuerId:record.code,period:row.period,column:j?'OP QoQ':'OP YoY',expected:value.expected,actual,sourceSha256:record.sourceSha256,...metadata,status:'PASS'});}
 }
}
expect(errors).toEqual([]);
const report={kind:'LOCAL_REAL_MARKET_PYTHON_DECIMAL_TO_BROWSER_DOM',scope:'Independent source quarter values and correct missingness compared with actual rendered OP YoY/QoQ columns; does not claim native Windows, public direct hosting, or original Android raw fields for these derived columns.',checkedAt:new Date().toISOString(),checks:checks.length,passed:checks.length,failed:0,pageErrors:errors,records:checks};
writeFileSync('/workspace/kr4-pc-work/quarter-op-independent-result.json',JSON.stringify(report,null,2));console.log(JSON.stringify({checks:checks.length,passed:checks.length,failed:0,pageErrors:errors.length}));
await browser.close();
})().catch(e=>{console.error(e);process.exit(1)});
